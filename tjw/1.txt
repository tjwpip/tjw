'随机生成身份证号码

'身份证号码各位数字的含义如下:
'1）前1、2位数字表示：所在省份的代码；
'（2）第3、4位数字表示：所在城市的代码；
'（3）第5、6位数字表示：所在区县的代码；
'（4）第7~14位数字表示：出生年、月、日；
'（5）第15、16位数字表示：所在地的派出所的代码；
'（6）第17位数字表示性别：奇数表示男性，偶数表示女性；
'（7）第18位数字是校检码：也有的说是个人信息码，一般是随计算机的随机产生，
'用来检验身份证的正确性。校检码可以是0~9的数字，有时也用x表示。


Dim collDiQu As New Collection           '归属地区       身份证前六位
Dim GuishudiA() As String

'生成身份证号
'520201 19850106 521 2
Public Function RndSFZ18() As String
    Dim lngSFZ As String

    '随机生成17位数
    lngSFZ = rndLeft1_6 & "" & RndYearMonthDay & "" & RndSn
    '生成最后一位验证码
    lngSFZ = lngSFZ & "" & GetYzm(lngSFZ)

    RndSFZ18 = lngSFZ & String(18 - Len(lngSFZ), "Z")

End Function


'验证身份证号的函数，参数ID是身份证号     身份证验证
Public Function SFZyz(ByVal id As String) As String
Dim sReturn As String                                                                               '返回值，返回为空表示成功，其它表示失败
Dim CsDate As String, CsYear As String, CsMonth As String, CsDay As String                          '出生日期
Dim lDay As String                                                                                  '临时变量


    BlnYZ = True
    '身份证位数验证
    If Len(id) <> 18 Then
        sReturn = sReturn & vbCrLf & "位数不对"
    End If

    '验证码 验证
    If GetYzm(id) <> Right(id, 1) Then
       sReturn = sReturn & vbCrLf & "验证码不对"
    End If

    '生日验证
    CsDate = Mid(id, 7, 8)
    CsYear = Left(CsDate, 4): CsMonth = Mid(CsDate, 5, 2): CsDay = Right(CsDate, 2)
    If CsMonth <= 0 Or CsMonth > 12 Then
        sReturn = sReturn & vbCrLf & "出生月不对"
    End If
    Select Case Val(CsMonth)
        Case 1, 3, 5, 6, 7, 8, 10, 12: lDay = 31
        Case 2
            If CsYear Mod 4 = 0 And CsYear Mod 100 <> 0 Or CsYear Mod 400 = 0 Then
                lDay = 29
            Else
                lDay = 28
            End If
        Case 4, 6, 9, 11: lDay = 30
    End Select
    If CsDay <= 0 Or CsDay > lDay Then
        sReturn = sReturn & vbCrLf & "出生日有误"
    End If


    SFZyz = sReturn
End Function

'获得归属地
Public Function GetDiqu(Optional ByVal id As String)
    Dim sReturn As String

    If Len(id) >= 6 Then
        sReturn = collDiQu(Left(id, 6))
    Else
        sReturn = collDiQu("0")
    End If

    GetDiqu = sReturn
End Function













'获得验证码 验证
Private Function GetYzm(ByVal id As String) As String
Dim Y As Integer, s As Long, Y1 As String
    Dim Jqyz() As String                        '加权因子
    Dim Yzm() As String                         '验证码
    Dim I As Integer

    Yzm = Split("1,0,X,9,8,7,6,5,4,3,2", ",")                       '验证码
    Jqyz = Split("7,9,10,5,8,4,2,1,6,3,7,9,10,5,8,4,2", ",")        '加权因子
'    For I = 0 To 16                                                 '加权因子可以这样求
'       Jqyz(I) = 2 ^ (17 - I) Mod 11
'    Next

    '对前17位数字的权求和
    For I = 0 To 16
        s = s + Val(Mid(id, I + 1, 1)) * Jqyz(I)
    Next
    Y = s Mod 11                                '对S取模
    Y1 = Yzm(Y)

    GetYzm = Y1
End Function

'随机生成前六位
Private Function rndLeft1_6() As String
    Dim s As String
    Dim l As Long

    Randomize
    l = Int((collDiQu.Count * Rnd) + 1)

    s = Val(GuishudiA(l))
    rndLeft1_6 = s
End Function


'生成年月日

Private Function RndYearMonthDay() As String
    Dim strYear As String, strMonth As String, strDay As String
    Dim lngYear As Long
    Dim l As Long

    lngYear = Year(Date)
    strYear = Int(Rnd * 100) + (lngYear - 100)

    strMonth = Int(Rnd * 12) + 1
    Select Case strMonth
        Case "1", "3", "5", "7", "8", "10", "12"
            l = 31
        Case "2"
            If (lngYear / 4 = 0 And lngYear / 100 <> 0) Or lngYear / 400 = 0 Then
                l = 29
            Else
                l = 28
            End If

        Case "4", "6", "9", "11"
            l = 30
        Case Else
            l = 28
    End Select

    strDay = Int(Rnd * l) + 1


    RndYearMonthDay = Format(strYear, "0000") & Format(strMonth, "00") & Format(strDay, "00")
End Function

'生成 同一天同一地点序列号
Private Function RndSn() As String
    Dim l As Long

    l = Rnd * 999 + 1

    RndSn = Format(l, "000")

End Function
